שלב 1 — הוסף Apps Script לגיליון

  1. פתח את הגיליון ← תוספות ← Apps Script
  2. מחק את הקוד הקיים והדבק את הקוד למטה
  3. שנה את SECRET_KEY לסיסמה שתבחר
  4. שנה את SHEET_NAME לשם הגיליון (לרוב גיליון1)
  5. שמור ← פרס (Deploy) ← New deployment
  6. Type: Web app | Execute as: Me | Access: Anyone
  7. העתק את כתובת ה-Web app URL
/* ═══ הגדרות — שנה כאן ═══ */
const SECRET_KEY = 'YOUR_SECRET_HERE'; // ← שנה לסיסמה שלך
const SHEET_NAME = 'גיליון1';          // ← שם הגיליון

/* ═══ קריאת נתונים (GET) ═══ */
function doGet(e) {
  if (e.parameter.key !== SECRET_KEY)
    return json({ error: 'Unauthorized' });
  const sheet = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  const data  = sheet.getDataRange().getValues();
  const hdrs  = data[0].map(h => h.toString().trim());
  const rows  = data.slice(1)
    .filter(r => r[0])
    .map(r => Object.fromEntries(hdrs.map((h, i) => [h, r[i]])));
  return json({ data: rows });
}

/* ═══ עדכון כמויות (POST) ═══ */
function doPost(e) {
  const params = JSON.parse(e.postData.contents);
  if (params.key !== SECRET_KEY)
    return json({ error: 'Unauthorized' });
  const sheet   = SpreadsheetApp.getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);
  const data    = sheet.getDataRange().getValues();
  const hdrs    = data[0].map(h => h.toString().trim());
  const idCol   = hdrs.indexOf('ID');
  const stkCol  = hdrs.indexOf('כמות_במלאי') + 1;
  let updated   = 0;
  for (const upd of params.updates) {
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][idCol]) === String(upd.id)) {
        sheet.getRange(i + 1, stkCol).setValue(upd.quantity);
        updated++;
        break;
      }
    }
  }
  return json({ success: true, updated });
}

function json(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
